## 📞 REFERENCE GUIDE

### Import Pattern
```jsx
import {
  Settings,
  Link as LinkIcon,
  MapPin,
  ImageIcon,
  Play,
  BarChart3,
  // ... more icons
} from 'lucide-react';
```

### Button Pattern
```jsx
<button className="btn btn-primary">
  <Icon size={16} />
  Label
</button>
```

### Card Pattern
```jsx
<div className="card">
  <h3>Title</h3>
  <p>Content</p>
</div>
```

### Status Pattern
```jsx
<div className="status-success">
  <CheckCircle2 size={16} />
  Message
</div>
```
